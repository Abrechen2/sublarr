# Manual Subtitle Upload Implementation Plan (backend)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a user hand-drop a subtitle file for an episode or movie — the last daily-use gap vs Bazarr — through the full existing security chain, saved as a managed sidecar recorded in history with `source="manual"`.

**Architecture:** A pure `services/subtitle_upload.py` validates + sanitizes the uploaded bytes (content-detected format, not extension-trusted). A `save_manual_subtitle` service resolves the sidecar path, handles overwrite-via-trash, writes atomically (`atomic_write_bytes`, which already sets 0644 for media-server readability), guards with `is_safe_path`, and records the download row. Two thin routes in `routes/subtitles/upload.py` resolve the episode/movie video path (mirroring the existing sidecar-list routes) and drive the services. Reuses every existing primitive; adds no new dependency.

**Tech Stack:** Python 3.11, Flask, SQLAlchemy, existing `subtitle_sanitizer` / `providers.format_validator` / `security_utils` / `utils.atomic_write` / `db.providers` helpers, pytest.

## Global Constraints

- Ruff `line-length=100`, `target-version="py311"`. Run `ruff check .` + `ruff format --check .` on the whole `backend/` dir before commit.
- **`is_safe_path()` is mandatory** on the resolved write path (import from `security_utils`) — this is a file-write endpoint.
- Content format is **detected from bytes**, never trusted from the filename extension.
- No new env vars, no new dependency.
- Tests: `python -m pytest --tb=short -q -n auto --ignore=tests/performance`.
- Conventional Commits.

## Scope note

**This plan covers raw subtitle files** (`.srt/.ass/.ssa/.vtt/.sub`). **Archive uploads (`.zip`) are rejected** in this plan with a clear 415 message ("extract the subtitle and upload the file directly"); zip-upload support (via the existing `archive_utils.extract_subtitles_from_zip`) is a fast follow-on plan. The **frontend** drag-&-drop + dialog in the EpisodeMenu is a separate follow-on; these backend endpoints are immediately usable via multipart POST.

```python
"""Validation + sanitization for manually-uploaded subtitle files.

Pure (no Flask/DB/disk). Format is detected from CONTENT, never trusted from
the filename extension. Archives are rejected here (raw files only).
"""

from __future__ import annotations

import os

ALLOWED_UPLOAD_EXTS: frozenset[str] = frozenset({"srt", "ass", "ssa", "vtt", "sub"})
MAX_UPLOAD_BYTES: int = 5 * 1024 * 1024
_ARCHIVE_EXTS: frozenset[str] = frozenset({"zip", "rar", "7z", "gz", "tar"})


class UploadError(Exception):
    """Raised when an uploaded subtitle is rejected. Carries an HTTP status."""

    def __init__(self, status: int, message: str):
        super().__init__(message)
        self.status = status
        self.message = message


def _ext_of(filename: str) -> str:
    return os.path.splitext(filename or "")[1].lstrip(".").lower()


def prepare_upload(filename: str, raw: bytes) -> tuple[bytes, str]:
    """Validate + sanitize an uploaded subtitle. Returns (content, detected_ext).

    Raises UploadError(status, message) on any rejection.
    """
    ext = _ext_of(filename)
    if ext in _ARCHIVE_EXTS:
        raise UploadError(
            415,
            "Archive uploads are not supported yet — extract the subtitle and "
            "upload the .srt/.ass/.vtt file directly.",
        )
    if ext not in ALLOWED_UPLOAD_EXTS:
        raise UploadError(415, f"Unsupported subtitle type: .{ext or '?'}")
    if not raw:
        raise UploadError(422, "Uploaded file is empty.")
    if len(raw) > MAX_UPLOAD_BYTES:
        raise UploadError(
            413, f"File too large: {len(raw) // 1024} KB > {MAX_UPLOAD_BYTES // 1024} KB limit"
        )

    from providers.format_validator import detect_format_from_content
    from subtitle_sanitizer import sanitize_subtitle, validate_content_type

    fmt = detect_format_from_content(raw)
    detected = getattr(fmt, "value", "unknown")
    if detected == "unknown" or not validate_content_type(raw, fmt):
        raise UploadError(422, "File content is not a recognised text subtitle.")

    try:
        content = sanitize_subtitle(raw, fmt)
    except ValueError as e:
        raise UploadError(422, f"Subtitle failed the security check: {e}") from e

    return content, detected
```

Adjust the `validate_content_type` / `detect_format_from_content` call form if Step-3 reading
shows a different signature (e.g. a string fmt instead of the enum) — keep the behavior
(content-detected, bitmap/binary → 422).

```python
def build_sidecar_path(video_path: str, language: str, modifier: str | None, ext: str) -> str:
    base, _ = os.path.splitext(video_path)
    parts = [base, language]
    if modifier:
        parts.append(modifier)
    parts.append(ext)
    return ".".join(parts)


def save_manual_subtitle(
    video_path: str,
    content: bytes,
    ext: str,
    language: str,
    modifier: str | None,
    overwrite: bool,
    media_path: str,
) -> str:
    """Write a validated manual subtitle as a sidecar and record it. Returns the path."""
    import security_utils
    from routes.subtitles.helpers import scan_subtitle_sidecars
    from utils.atomic_write import atomic_write_bytes

    out_path = build_sidecar_path(video_path, language, modifier, ext)

    if not security_utils.is_safe_path(out_path, media_path):
        raise UploadError(400, "Resolved subtitle path is outside the media directory.")

    existing = None
    for side in scan_subtitle_sidecars(video_path):
        if side.get("language") == language and side.get("modifier") == modifier:
            existing = side["path"]
            break
    if existing and not overwrite:
        raise UploadError(409, f"A {language} subtitle already exists — set overwrite to replace.")
    if existing and overwrite:
        try:
            from services.sidecar_trash import get_batch_dir, get_trash_root, trash_sidecar

            batch_dir = get_batch_dir(media_path, "manual-upload")
            os.makedirs(batch_dir, exist_ok=True)
            trash_sidecar(existing, media_path, batch_dir)
        except Exception:
            # If trashing the old file fails, the atomic overwrite below still
            # replaces it; do not abort the upload.
            pass

    atomic_write_bytes(out_path, content)

    try:
        from db.providers import record_subtitle_download

        record_subtitle_download(
            provider_name="manual",
            subtitle_id=f"manual:{os.path.basename(out_path)}",
            language=language,
            fmt=ext,
            file_path=out_path,
            score=0,
            source="manual",
        )
    except Exception:
        # History recording must never lose the written file.
        pass

    return out_path
```

Note: reading via `scan_subtitle_sidecars` reuses the canonical parser so "same language +
modifier" matches how the rest of the app sees sidecars. The `modifier` from
`scan_subtitle_sidecars` is absent (None) for plain subs — the comparison handles both.

```python
"""Manual subtitle upload endpoints for episodes and standalone movies."""

from __future__ import annotations

import logging
import os

from flask import jsonify, request

from config import get_settings, map_path
from routes.subtitles import bp
from services.subtitle_upload import UploadError, prepare_upload, save_manual_subtitle

logger = logging.getLogger(__name__)


def _modifier_from_form() -> str | None:
    if request.form.get("hi", "").lower() in ("1", "true", "yes", "on"):
        return "hi"
    if request.form.get("forced", "").lower() in ("1", "true", "yes", "on"):
        return "forced"
    mod = (request.form.get("modifier") or "").strip().lower()
    return mod or None


def _do_upload(video_path: str):
    upload = request.files.get("file")
    if upload is None or not upload.filename:
        return jsonify({"error": "No file provided"}), 400
    language = (request.form.get("language") or "").strip().lower()
    if not language:
        return jsonify({"error": "Missing 'language'"}), 400
    overwrite = request.form.get("overwrite", "").lower() in ("1", "true", "yes", "on")
    modifier = _modifier_from_form()

    try:
        content, ext = prepare_upload(upload.filename, upload.read())
        media_path = getattr(get_settings(), "media_path", "/media")
        saved = save_manual_subtitle(
            video_path, content, ext, language, modifier, overwrite, media_path
        )
    except UploadError as e:
        return jsonify({"error": e.message}), e.status

    return jsonify({"saved_path": saved, "language": language, "format": ext}), 201


@bp.route("/library/episodes/<int:ep_id>/subtitles/upload", methods=["POST"])
def upload_episode_subtitle(ep_id: int):
    from sonarr_client import get_sonarr_client

    client = get_sonarr_client()
    if client is None:
        return jsonify({"error": "Sonarr not configured"}), 503
    raw_path = client.get_episode_file_path(ep_id)
    if not raw_path:
        return jsonify({"error": "Episode has no video file"}), 404
    video_path = map_path(raw_path)
    if not os.path.exists(video_path):
        return jsonify({"error": "Video file not found"}), 404
    return _do_upload(video_path)


@bp.route("/library/movies/<int:movie_id>/subtitles/upload", methods=["POST"])
def upload_movie_subtitle(movie_id: int):
    from db.standalone import get_standalone_movies

    movie = get_standalone_movies(movie_id)
    if movie is None:
        return jsonify({"error": "Movie not found"}), 404
    file_path = movie.get("file_path") if isinstance(movie, dict) else getattr(movie, "file_path", None)
    if not file_path or not os.path.exists(file_path):
        return jsonify({"error": "Video file not found"}), 404
    return _do_upload(file_path)
```
